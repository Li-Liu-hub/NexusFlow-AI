# 全 Go 切换手册（feat/go-rewrite @ fe30627）

> 前置：5 个 Linux 二进制已在本仓库 CI 产物 / 本机 `go-bin/`：
> gateway / platform-backend / terminal-agent / kernel / orchestrator（合计 ~110MB）。
> 本手册在服务器上执行，将 Java 栈整体替换为 Go 栈，**零 JVM**。

## 0. 上传二进制

```bash
# 本机（Git Bash）
GOBIN=/opt/agent-platform/go-bin  # 服务器目录
for f in gateway platform-backend terminal-agent kernel orchestrator; do
  python /d/env/upload_file.py "$(cygpath -w /tmp/go-bin/$f)" "/home/ubuntu/go-$f"
done
```

## 1. 服务器上切换（一次执行）

```bash
sudo mkdir -p /opt/agent-platform/go-bin
sudo mv /home/ubuntu/go-* /opt/agent-platform/go-bin/
sudo chmod +x /opt/agent-platform/go-bin/*
sudo bash -c 'set -a; . /opt/agent-platform/compose/.env; set +a; GATEWAY_SK=${GATEWAY_SK:-7c1f9a3e5d0b2846af31c7e9b0d45268}

# 停 5 个 Java 容器（保留 mysql/postgres/redis/nginx；rabbitmq/nacos 稍后一并下线）
docker rm -f ap-orchestrator ap-digital-human ap-agent ap-backend ap-gateway

NET=agent-platform_apnet
BIN=/opt/agent-platform/go-bin
COMMON_ENV="-e MYSQL_DSN=${MYSQL_APP_USER}:${MYSQL_APP_PW}@tcp(mysql:3306)/api_db?parseTime=true&loc=Local&charset=utf8mb4&timeout=5s -e REDIS_ADDR=redis:6379 -e REDIS_PW=${REDIS_PW}"

# 1) 网关（占 172.28.0.11，nginx 的 upstream 名不变）
docker run -d --name ap-gateway --restart unless-stopped --network $NET --network-alias gateway \
  --ip 172.28.0.11 --memory 96m -v $BIN:/app:ro --entrypoint /app/gateway nginx:1.27-alpine \
  $COMMON_ENV \
  -e BACKEND_URL=http://backend:7529 -e AGENT_URL=http://agent:8124 \
  -e KERNEL_URL=http://kernel:8126 -e ORCHESTRATOR_URL=http://orchestrator:8125 \
  -e MOCK_URL=http://127.0.0.1:8123 -e PORTAL_ACCESS_KEY=aismp-console \
  -e COUNT_URL=http://backend:7529/api/inner/userInterfaceInfo/invokeCount \
  -e RATE_PER_MINUTE=600

# 2) platform-backend（:7529，占 172.28.0.12）
docker run -d --name ap-backend --restart unless-stopped --network $NET --network-alias backend \
  --ip 172.28.0.12 --memory 96m -v $BIN:/app:ro --entrypoint /app/platform-backend nginx:1.27-alpine \
  -e HTTP_ADDR=:7529 -e MYSQL_DSN="${MYSQL_APP_USER}:${MYSQL_APP_PW}@tcp(mysql:3306)/api_db?parseTime=true&loc=Local&charset=utf8mb4&timeout=5s" \
  -e GATEWAY_BASE_URL=http://gateway:8090 -e DIGITAL_HUMAN_BASE_URL=http://kernel:8126 \
  -e AISMP_CLIENT_ACCESS_KEY=aismp-console -e AISMP_CLIENT_SECRET_KEY=$GATEWAY_SK

# 3) terminal-agent（:8124）
docker run -d --name ap-agent --restart unless-stopped --network $NET --network-alias agent \
  --memory 96m -v $BIN:/app:ro --entrypoint /app/terminal-agent nginx:1.27-alpine \
  -e HTTP_ADDR=:8124 -e MYSQL_DSN="${MYSQL_APP_USER}:${MYSQL_APP_PW}@tcp(mysql:3306)/api_db?parseTime=true&loc=Local&charset=utf8mb4&timeout=5s"

# 4) kernel（:8126）—— env 从旧 ap-digital-human 继承（LLM key / PG / terminal_db 凭据）
#    旧容器的环境变量先用 docker inspect 备份（见 §2）
docker run -d --name ap-kernel --restart unless-stopped --network $NET --network-alias kernel \
  --memory 128m -v $BIN:/app:ro --entrypoint /app/kernel nginx:1.27-alpine \
  -e HTTP_ADDR=:8126 \
  -e TERMINAL_DSN="${KERNEL_DB_USER:-${MYSQL_APP_USER}}:${KERNEL_DB_PW:-${MYSQL_APP_PW}}@tcp(mysql:3306)/terminal_db?parseTime=true&loc=Local&charset=utf8mb4&timeout=5s" \
  -e API_DSN="${MYSQL_APP_USER}:${MYSQL_APP_PW}@tcp(mysql:3306)/api_db?parseTime=true&loc=Local&charset=utf8mb4&timeout=5s" \
  -e PG_DSN="postgres://${PG_USER}:${PG_PASSWORD}@postgres:5432/terminal_kb?sslmode=disable" \
  -e LLM_BASE_URL=https://api.xiaomimimo.com -e MIMO_API_KEY=$MIMO_API_KEY \
  -e LLM_MODEL=${LLM_MODEL:-mimo-v2.6-flash} -e SILICONFLOW_API_KEY=$SILICONFLOW_API_KEY \
  -e BACKEND_BASE_URL=http://backend:7529/api -e AGENT_BASE_URL=http://agent:8124 \
  -e GATEWAY_BASE_URL=http://gateway:8090 -e GATEWAY_ACCESS_KEY=aismp-console -e GATEWAY_SECRET_KEY=$GATEWAY_SK

# 5) orchestrator（:8125）
docker run -d --name ap-orchestrator --restart unless-stopped --network $NET --network-alias orchestrator \
  --memory 64m -v $BIN:/app:ro --entrypoint /app/orchestrator nginx:1.27-alpine \
  -e AGENT_BASE_URL=http://agent:8124 -e KERNEL_BASE_URL=http://kernel:8126 \
  -e KERNEL_USER=svc_orchestrator -e KERNEL_PASS=${KERNEL_SVC_PW:-svc-orch-2026} \
  -e KB_AGENT_CODE=${KB_AGENT_CODE:-kb_assistant}

# 6) 下线不再需要的 JVM 中间件（Nacos 本身是 JVM；RabbitMQ 无消费者）
docker rm -f ap-nacos ap-rabbitmq
'
```

## 2. 旧容器环境变量备份（切换前执行！）

```bash
for c in ap-digital-human ap-agent ap-backend ap-orchestrator; do
  sudo docker inspect $c --format '{{range .Config.Env}}{{println .}}{{end}}' > ~/env-$c.bak
done
# 关键变量：MIMO_API_KEY / SILICONFLOW_API_KEY / PG_PASSWORD / KERNEL 的 DB 凭据 /
#           KB_AGENT_CODE / KERNEL_SVC_PW
```
